cmake_minimum_required(VERSION 3.14)
project(LinuxWebViewConfiguration LANGUAGES C CXX)

# Exercise Linux configuration policy without Linux SDKs or a cross compiler.
set(CMAKE_SYSTEM_NAME Linux)
set(APPLE FALSE)
set(WIN32 FALSE)
set(UNIX TRUE)
set(IPLUG2_CMAKE_DIR "${IPLUG2_DIR}/Scripts/cmake")

if(TEST_MODE STREQUAL "module")
  include("${IPLUG2_CMAKE_DIR}/WebView.cmake")
elseif(TEST_MODE MATCHES "^examples")
  if(TEST_MODE STREQUAL "examples-darwin")
    set(CMAKE_SYSTEM_NAME Darwin)
    set(APPLE TRUE)
  endif()
  function(add_subdirectory directory)
    set_property(GLOBAL APPEND PROPERTY test_examples "${directory}")
  endfunction()
  include("${IPLUG2_DIR}/Examples/CMakeLists.txt")
  get_property(examples GLOBAL PROPERTY test_examples)
  foreach(example IPlugOSCEditor IPlugWebUI IPlugP5js IPlugSvelteUI IPlugReaperExtensionWebUI)
    if(TEST_MODE STREQUAL "examples-darwin" AND NOT example IN_LIST examples)
      message(FATAL_ERROR "Supported example was excluded: ${example}")
    elseif(TEST_MODE STREQUAL "examples" AND example IN_LIST examples)
      message(FATAL_ERROR "Unsupported example was included: ${example}")
    endif()
  endforeach()
  foreach(example IPlugEffect IPlugInstrument IPlugReaperExtension IPlugReaperPlugin)
    if(NOT example IN_LIST examples)
      message(FATAL_ERROR "Supported example was excluded: ${example}")
    endif()
  endforeach()
else()
  include("${IPLUG2_CMAKE_DIR}/IPlug.cmake")
  if(NOT TARGET iPlug2::IPlug OR TARGET iPlug2::Extras::IWebViewControl)
    message(FATAL_ERROR "Linux core/WebView target availability is incorrect")
  endif()
  include("${IPLUG2_CMAKE_DIR}/IPlugPlugin.cmake")
  if(TEST_MODE STREQUAL "plugin")
    iplug_add_plugin(WebViewProbe SOURCES "${CMAKE_CURRENT_LIST_FILE}" UI WEBVIEW)
  elseif(TEST_MODE STREQUAL "embedded")
    iplug_add_plugin(WebViewProbe SOURCES "${CMAKE_CURRENT_LIST_FILE}"
      LINK iPlug2::Extras::IWebViewControl)
  elseif(TEST_MODE STREQUAL "direct")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/probe.cpp" "int main() { return 0; }\n")
    add_executable(WebViewProbe "${CMAKE_CURRENT_BINARY_DIR}/probe.cpp")
    target_link_libraries(WebViewProbe PRIVATE iPlug2::Extras::IWebViewControl)
  elseif(NOT TEST_MODE STREQUAL "core")
    message(FATAL_ERROR "Unknown test mode: ${TEST_MODE}")
  endif()
endif()
