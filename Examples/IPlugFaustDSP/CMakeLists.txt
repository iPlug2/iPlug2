cmake_minimum_required(VERSION 3.14)
project(IPlugFaustDSP VERSION 1.0.0)

# Discover IPLUG2_DIR if not already set (for independent builds)
if(NOT DEFINED IPLUG2_DIR)
  set(IPLUG2_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "iPlug2 root directory")
endif()

# Include iPlug2 CMake configuration
include(${IPLUG2_DIR}/iPlug2.cmake)

find_package(iPlug2 REQUIRED)

# Finds an installed Faust (set FAUST_DIR if it isn't found)
include(${IPLUG2_CMAKE_DIR}/Faust.cmake)

if(NOT IPLUG2_FAUST_SUPPORTED)
  message(STATUS "Skipping ${PROJECT_NAME}: Faust was not found")
  return()
endif()

iplug_add_plugin(${PROJECT_NAME}
  FORMATS DESKTOP
  SOURCES
    IPlugFaustDSP.cpp
    IPlugFaustDSP.h
    IPlugFaustDSP.dsp
    resources/resource.h
  RESOURCES
    resources/fonts/Roboto-Regular.ttf
)

# By default, Debug builds compile IPlugFaustDSP.dsp at runtime with the libfaust interpreter (and recompile it when it
# changes), other builds use C++ generated by the faust compiler.
# With IPLUGFAUSTDSP_SHIP_INTERPRETER, all builds use the interpreter, and the .dsp file and the Faust libraries are
# copied into the bundle, so that the DSP is compiled when the plug-in loads, also on other computers and on iOS.
# The class name is used with FAUST_BLOCK() in IPlugFaustDSP.h
option(IPLUGFAUSTDSP_SHIP_INTERPRETER "Use the Faust interpreter in all configurations, and bundle the .dsp file and Faust libraries" OFF)

if(IPLUGFAUSTDSP_SHIP_INTERPRETER)
  iplug_add_faust(${PROJECT_NAME}
    DSP IPlugFaustDSP.dsp
    CLASS FaustEffect
    MODE INTERPRETER
    BUNDLE
  )
else()
  iplug_add_faust(${PROJECT_NAME}
    DSP IPlugFaustDSP.dsp
    CLASS FaustEffect
  )
endif()
